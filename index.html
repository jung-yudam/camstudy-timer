    function initAudio() {

      if (!audioCtx) {

        audioCtx = new (window.AudioContext || window.webkitAudioContext)();

      }

      if (audioCtx.state === "suspended") {

        audioCtx.resume();

      }

    }


    function playChime(type) {

      if (!soundEnabled) return;

      initAudio();


      const now = audioCtx.currentTime;

      const notes =

        type === "study"

          ? [523.25, 659.25, 783.99]

          : [783.99, 659.25, 523.25];


      notes.forEach((freq, idx) => {

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();


        osc.type = "sine";

        osc.frequency.setValueAtTime(freq, now + idx * 0.18);


        gain.gain.setValueAtTime(0.001, now + idx * 0.18);

        gain.gain.exponentialRampToValueAtTime(0.24, now + idx * 0.18 + 0.03);

        gain.gain.exponentialRampToValueAtTime(0.0001, now + idx * 0.18 + 0.7);


        osc.connect(gain);

        gain.connect(audioCtx.destination);


        osc.start(now + idx * 0.18);

        osc.stop(now + idx * 0.18 + 0.75);

      });

    }

    function getPeriodNumber(hour) {

      if (hour >= 9 && hour < 12) return hour - 8;

      if (hour >= 13 && hour < 18) return hour - 9;

      if (hour >= 19 && hour < 23) return hour - 10;

      if (hour === 23) return 13;

      if (hour >= 0 && hour < 9) return hour + 14;

      return null;

    }


    function pad(num) {

      return String(num).padStart(2, "0");

    }


    function highlightSchedule(rangeKey) {

      document.querySelectorAll(".schedule-item").forEach((el) => {

      });

    }


    function updateBreakTip(isBreak, minute) {

      if (!isBreak) {

        breakTipBox.classList.remove("show");

        lastTipIndex = -1;

        return;

      }


      // 50~59분 사이 1분마다 안내 문구가 순서대로 전환됨

      const tipIndex = (minute - 50) % BREAK_TIPS.length;

      if (tipIndex !== lastTipIndex) {

        const tip = BREAK_TIPS[tipIndex];

        breakTipIcon.textContent = tip.icon;

        breakTipText.textContent = tip.text;

        // 문구가 바뀔 때 부드러운 전환 애니메이션 재실행

        breakTipBox.classList.remove("show");

        void breakTipBox.offsetWidth;

        breakTipBox.classList.add("show");

        lastTipIndex = tipIndex;

      } else {

        breakTipBox.classList.add("show");

      }

    }


    function updateRunner(h, m, s) {

      const startSec = 9 * 3600;

      const endSec = 23 * 3600;

      const totalSpan = endSec - startSec;

      const nowSec = h * 3600 + m * 60 + s;


      const fill = document.getElementById("trackFill");

      const statusEl = document.getElementById("runnerStatus");


      if (h >= 9 && h < 23) {

        const pct = Math.min(100, Math.max(0, ((nowSec - startSec) / totalSpan) * 100));

        runnerAvatar.style.left = `${pct}%`;

        fill.style.width = `${pct}%`;

        runnerAvatar.classList.add("running");

        statusEl.textContent = `09:00 ~ 23:00 레이스 진행 중 (${pct.toFixed(1)}%)`;

      } else if (h >= 23) {

        runnerAvatar.style.left = "100%";

        fill.style.width = "100%";

        runnerAvatar.classList.remove("running");

        statusEl.textContent = "23:00 정규 레이스 완주! (심야 뽀모도로 이어달리기 중)";

      } else {

        runnerAvatar.style.left = "0%";

        fill.style.width = "0%";

        runnerAvatar.classList.remove("running");

        statusEl.textContent = "09:00 출발 대기 중";

      }

    }


    function updateTimer() {

      const now = new Date();

      const h = now.getHours();

      const m = now.getMinutes();

      const s = now.getSeconds();


      document.getElementById("currentClock").textContent =

        `현재 시각 ${pad(h)}:${pad(m)}:${pad(s)}`;


      const isDaySky = h >= 6 && h < 18;

      document.getElementById("skyIcon").textContent = isDaySky ? "☀️" : "🌙";

      const bannerTitle = document.getElementById("bannerTitle");

      const statusBadge = document.getElementById("statusBadge");

      const sessionInfo = document.getElementById("sessionInfo");

      const timerDisplay = document.getElementById("timerDisplay");

      const progressBar = document.getElementById("progressBar");

        specialBanner.classList.add("show", "finish-banner");

        bannerTitle.textContent = "오늘 하루도 수고 많았어요! ✨";

      } else {

        specialBanner.classList.remove("show");

      }


      let stateKey = "";

      let remainingSec = 0;

      let totalSec = 0;

      let soundType = "break";

      let isBreakTime = false;


      if (h === 12) {

        stateKey = "lunch";

        soundType = "meal";

        remainingSec = (59 - m) * 60 + (60 - s);

        totalSec = 3600;


        highlightSchedule("12-13");

        specialBanner.classList.remove("finish-banner");

        specialBanner.classList.add("show");

        bannerTitle.textContent = "🍱 지금은 점심시간입니다!";

        bannerSub.textContent = "맛있게 식사하고 쉬세요 · 13:00에 4교시가 시작됩니다.";


        document.body.style.backgroundColor = "var(--bg-meal)";

        statusBadge.textContent = "🍱 점심 시간";

        statusBadge.style.background = "rgba(251, 191, 36, 0.16)";

        statusBadge.style.color = "var(--accent-meal)";

        sessionInfo.textContent = "13:00 ~ 13:50 4교시 집중 예정";

        progressBar.style.backgroundColor = "var(--accent-meal)";

      } else if (h === 18) {

        stateKey = "dinner";

        soundType = "meal";

        remainingSec = (59 - m) * 60 + (60 - s);

        totalSec = 3600;


        highlightSchedule("18-19");

        specialBanner.classList.remove("finish-banner");

        specialBanner.classList.add("show");

        bannerTitle.textContent = "🍽️ 지금은 저녁시간입니다!";

        bannerSub.textContent = "든든하게 저녁 챙겨드세요 · 19:00에 9교시가 시작됩니다.";


        document.body.style.backgroundColor = "var(--bg-meal)";

        statusBadge.textContent = "🍽️ 저녁 시간";

        statusBadge.style.background = "rgba(251, 191, 36, 0.16)";

        statusBadge.style.color = "var(--accent-meal)";

        sessionInfo.textContent = "19:00 ~ 19:50 9교시 집중 예정";

        progressBar.style.backgroundColor = "var(--accent-meal)";

      } else {

        const period = getPeriodNumber(h);

        const isNightSession = h >= 23 || h < 9;

        const nextHour = (h + 1) % 24;


        if (h >= 9 && h < 12) highlightSchedule("09-12");

        else if (h >= 13 && h < 18) highlightSchedule("13-18");

        else if (h >= 19 && h < 23) highlightSchedule("19-23");

        else highlightSchedule("23-09");


        if (m < 50) {

          stateKey = `study-${h}`;

          soundType = "study";

          remainingSec = (49 - m) * 60 + (60 - s);

          totalSec = 50 * 60;


          document.body.style.backgroundColor = isNightSession

            ? "var(--bg-night)"

            : h >= 19

            ? "var(--bg-evening)"

            : "var(--bg-day)";


          statusBadge.textContent = isNightSession

            ? `🌙 심야 ${period}교시 집중 시간`

            : `🔥 ${period}교시 집중 시간`;

          statusBadge.style.background = isNightSession

            ? "rgba(167, 139, 250, 0.16)"

            : "rgba(56, 189, 248, 0.16)";

          statusBadge.style.color = isNightSession

            ? "var(--accent-night)"

            : "var(--accent-study)";

          sessionInfo.textContent = `${pad(h)}:00 ~ ${pad(h)}:50 집중 진행 중`;

          progressBar.style.backgroundColor = isNightSession

            ? "var(--accent-night)"

            : "var(--accent-study)";

        } else {

          isBreakTime = true;

          stateKey = `break-${h}`;

          soundType = "break";

          remainingSec = (59 - m) * 60 + (60 - s);

          totalSec = 10 * 60;


          document.body.style.backgroundColor = "#11221c";

          statusBadge.textContent = `☕ ${period}교시 종료 · 휴식 시간`;

          statusBadge.style.background = "rgba(52, 211, 153, 0.16)";

          statusBadge.style.color = "var(--accent-break)";


          let nextText = "";

          if (h === 11) {

            nextText = "12:00부터 점심 시간입니다";

          } else if (h === 17) {

            nextText = "18:00부터 저녁 시간입니다";

          } else {

            const nextPeriod = getPeriodNumber(nextHour);

            nextText = `${pad(nextHour)}:00에 ${nextPeriod}교시가 시작됩니다`;

          }

          sessionInfo.textContent = nextText;

          progressBar.style.backgroundColor = "var(--accent-break)";

        }

      }


      updateBreakTip(isBreakTime, m);


      if (lastStateKey !== null && lastStateKey !== stateKey) {

        playChime(soundType);

      }

      lastStateKey = stateKey;


      const dispM = Math.floor(remainingSec / 60);

      const dispS = remainingSec % 60;

      const formattedTime = `${pad(dispM)}:${pad(dispS)}`;


      timerDisplay.textContent = formattedTime;


      const elapsed = totalSec - remainingSec;

      const pct = Math.min(100, Math.max(0, (elapsed / totalSec) * 100));

      progressBar.style.width = `${pct}%`;


      updateRunner(h, m, s);


      document.title = `(${formattedTime}) ${statusBadge.textContent}`;

    }


    updateTimer();

    setInterval(updateTimer, 250);
